function showMatches(data) { matchesBox.innerHTML = ""; const list = Array.isArray(data) ? data : data.matches || data.data || data.events || []; if (!Array.isArray(list) || list.length === 0) { matchStatus.textContent = "No matches found."; return; } matchStatus.textContent = list.length + " match(es)"; function cleanValue(value) { if (value === null || value === undefined) { return ""; } if (typeof value === "string" || typeof value === "number" || typeof value === "boolean") { return String(value); } if (Array.isArray(value)) { return value .map(cleanValue) .filter(Boolean) .join(" • "); } if (typeof value === "object") { // Common team/object formats if (value.name) return cleanValue(value.name); if (value.teamName) return cleanValue(value.teamName); if (value.shortName) return cleanValue(value.shortName); if (value.title) return cleanValue(value.title); return Object.values(value) .map(cleanValue) .filter(Boolean) .join(" • "); } return ""; } list.forEach((match) => { const card = document.createElement("article"); card.className = "match-card"; /* MATCH TITLE */ const title = document.createElement("h3"); let matchTitle = match.name || match.title || match.matchName; if (!matchTitle && match.team1 && match.team2) { matchTitle = cleanValue(match.team1) + " vs " + cleanValue(match.team2); } if (!matchTitle && match.homeTeam && match.awayTeam) { matchTitle = cleanValue(match.homeTeam) + " vs " + cleanValue(match.awayTeam); } title.textContent = cleanValue(matchTitle) || "Cricket Match"; card.appendChild(title); /* STATUS */ const statusText = match.status || match.matchStatus || match.state; if (statusText) { const status = document.createElement("p"); if (tournament) { const series = document.createElement("p"); series.textContent = "Series: " + cleanValue(tournament); card.appendChild(series); } /* VENUE */ if (match.venue) { const venue = document.createElement("p"); venue.textContent = "Venue: " + cleanValue(match.venue); card.appendChild(venue); } /* SCORE */ const score = match.score || match.scorecard || match.scores; if (score) { const scoreElement = document.createElement("p"); scoreElement.textContent = "Score: " + cleanValue(score); card.appendChild(scoreElement); } /* DATE / TIME */ const dateTime = match.date || match.dateTime || match.startTime || match.start; if (dateTime) { const time = document.createElement("p"); time.textContent = "Time: " + cleanValue(dateTime); card.appendChild(time); } matchesBox.appendChild(card); }); }
